cmake_minimum_required(VERSION 3.28)
project(AsyncModuleProbes LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)

set(ASYNC_MODULE_PROBE deduction CACHE STRING "Probe: deduction, asio or stdexec")
set_property(CACHE ASYNC_MODULE_PROBE PROPERTY STRINGS deduction asio stdexec)
option(ASYNC_MODULE_PROBE_USE_MODULES "Import the probe instead of including headers" ON)
set(ASYNC_MODULE_PROBE_INCLUDE_DIRECTORY "" CACHE PATH "Native dependency include directory")
if(NOT ASYNC_MODULE_PROBE MATCHES "^(deduction|asio|stdexec)$")
    message(FATAL_ERROR "Unknown probe: ${ASYNC_MODULE_PROBE}")
endif()
if(NOT ASYNC_MODULE_PROBE STREQUAL "deduction" AND NOT IS_DIRECTORY "${ASYNC_MODULE_PROBE_INCLUDE_DIRECTORY}")
    message(FATAL_ERROR "Set ASYNC_MODULE_PROBE_INCLUDE_DIRECTORY to the native dependency's include directory")
endif()

add_library(probe_options INTERFACE)
target_compile_features(probe_options INTERFACE cxx_std_23)
target_include_directories(probe_options INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
if(ASYNC_MODULE_PROBE_INCLUDE_DIRECTORY)
    target_include_directories(probe_options INTERFACE "${ASYNC_MODULE_PROBE_INCLUDE_DIRECTORY}")
endif()
if(MSVC)
    target_compile_options(probe_options INTERFACE /EHsc /Zc:__cplusplus /Zc:preprocessor /permissive- /bigobj)
endif()
if(ASYNC_MODULE_PROBE STREQUAL "asio")
    target_compile_definitions(probe_options INTERFACE ASIO_STANDALONE ASIO_SEPARATE_COMPILATION _WIN32_WINNT=0x0601)
endif()

if(ASYNC_MODULE_PROBE_USE_MODULES)
    add_library(probe_interface STATIC)
    target_sources(probe_interface PUBLIC FILE_SET CXX_MODULES FILES "${ASYNC_MODULE_PROBE}.cppm")
    target_link_libraries(probe_interface PUBLIC probe_options)
endif()

# Compile only: link/runtime behavior is covered by the library's regular sandbox.
add_library(probe_consumer OBJECT "${ASYNC_MODULE_PROBE}_consumer.cpp")
target_link_libraries(probe_consumer PRIVATE probe_options)
target_compile_definitions(probe_consumer PRIVATE PROBE_USE_MODULES=$<BOOL:${ASYNC_MODULE_PROBE_USE_MODULES}>)
set_target_properties(probe_consumer PROPERTIES CXX_EXTENSIONS OFF CXX_SCAN_FOR_MODULES "${ASYNC_MODULE_PROBE_USE_MODULES}")
if(ASYNC_MODULE_PROBE_USE_MODULES)
    target_link_libraries(probe_consumer PRIVATE probe_interface)
endif()
