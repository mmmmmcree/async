if(NOT LCF_ASYNC_STANDALONE_TESTS_IN_MAIN_BUILD)
    include(cmake/deferred_targets.cmake)
    return()
endif()
foreach(kind verify sandbox lessons benchmark)
    if(kind STREQUAL "benchmark")
        set(option BENCHMARKS)
    else()
        string(TOUPPER "${kind}" option)
    endif()
    if(LCF_ASYNC_STANDALONE_BUILD_${option})
        if(kind STREQUAL "benchmark")
            include(../../cmake/test_tools.cmake)
            async_test_dependency(STANDALONE BENCHMARK benchmark::benchmark benchmark googlebenchmark
                https://github.com/google/benchmark.git 192ef10025eb2c4cdd392bc502f0c852196baa48)
        endif()
        add_subdirectory(${kind})
    endif()
endforeach()
